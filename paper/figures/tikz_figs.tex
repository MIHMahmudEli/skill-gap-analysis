% ---- shared style (light, print-safe palette) ----
% Fills are deliberately pale so that value labels, CI whiskers and the thin
% bar outlines stay legible on top of them; series are separated by hue, not
% by saturation, and no bar carries a dark border.
\definecolor{sgBlue}{RGB}{147,179,224}    % demand / market / primary
\definecolor{sgTeal}{RGB}{126,208,178}    % lexical measurement
\definecolor{sgGreen}{RGB}{176,208,141}   % semantic measurement
\definecolor{sgOrange}{RGB}{245,197,133}  % lexical deficit
\definecolor{sgRed}{RGB}{238,165,156}     % semantic deficit / contrast
\definecolor{sgGrey}{RGB}{152,160,168}    % reference lines, whiskers
\definecolor{sgLight}{RGB}{233,235,237}   % grid
\definecolor{sgAxis}{RGB}{196,201,206}    % axis lines, ticks

\pgfplotsset{
  sg axis/.style={
    axis line style={sgAxis, line width=0.4pt},
    tick style={sgAxis, line width=0.4pt},
    label style={font=\footnotesize, color=black!72},
    tick label style={font=\scriptsize, color=black!62},
    legend style={
      font=\scriptsize,
      draw=none,
      fill=none,
      cells={anchor=west},
      inner sep=2pt,
      /tikz/every even column/.append style={column sep=5pt},
    },
    ymin=0,
    ymajorgrids=true,
    grid style={sgLight, line width=0.4pt},
    width=0.98\linewidth,
    height=0.72\linewidth,
    scale only axis,
  },
}
\tikzset{
  sg bar/.style={
    fill=#1,
    draw=#1!80!black,
    line width=0.35pt,
  },
}
